**Goal:** Stamp a version into the app and guarantee that a customer document never loses its header or footer at a page break, and never splits a paragraph across one.

**Order matters.** Task 3 fixes the path Mattias actually uses. Task 6 (browser print) is last because it is the hardest and the least used — see its note before starting it.

```python
# tests/harness.py
import contextlib
import functools
import http.server
import pathlib
import socketserver
import threading

ROOT = pathlib.Path(__file__).resolve().parents[1]
PORT = 8142
APP_PATH = "/crosscontrol-offertbyggare.html"

socketserver.TCPServer.allow_reuse_address = True


@contextlib.contextmanager
def serve():
    """Serve the repo root over localhost. file:// is refused by Playwright."""
    handler = functools.partial(http.server.SimpleHTTPRequestHandler, directory=str(ROOT))
    httpd = socketserver.TCPServer(("127.0.0.1", PORT), handler)
    threading.Thread(target=httpd.serve_forever, daemon=True).start()
    try:
        yield f"http://127.0.0.1:{PORT}{APP_PATH}"
    finally:
        httpd.shutdown()
        httpd.server_close()


@contextlib.contextmanager
def app_page(playwright, url):
    """Open the app and yield (page, alerts). Any alert() is captured, never swallowed."""
    browser = playwright.chromium.launch()
    page = browser.new_page()
    alerts = []
    page.on("dialog", lambda d: (alerts.append(d.message), d.accept()))
    page.goto(url)
    page.wait_for_function(
        "typeof addProductToCart === 'function'"
        " && typeof FLAT_PRODUCTS !== 'undefined' && FLAT_PRODUCTS.length > 0"
    )
    try:
        yield page, alerts
    finally:
        browser.close()


def build_long_quote(page, n_products=16):
    """Build a quote long enough to run past three pages."""
    page.fill("#custName", "Husco International")
    page.evaluate(
        "(n) => { FLAT_PRODUCTS.slice(0, n).forEach(p => addProductToCart(p.part_number)); }",
        n_products,
    )
    page.wait_for_timeout(400)


def download_quote_pdf(page, alerts, out_path):
    with page.expect_download(timeout=30000) as dl:
        page.click("#downloadPdfBtn")
    dl.value.save_as(str(out_path))
    assert not alerts, f"alert fired during export: {alerts}"
    return out_path


def page_texts(pdf_path):
    """Text of each PDF page, newlines flattened, as a list."""
    import fitz

    with fitz.open(str(pdf_path)) as doc:
        return [p.get_text().replace("\n", " ") for p in doc]
```

- [ ] **Step 7: ~~Stop the product table breaking inside a row~~ — WITHDRAWN 2026-08-25**

This step originally added `dontBreakRows: true` to the product table. **Do not do it.** A code review reproduced the consequence against the real export button: `dontBreakRows` is row-level `unbreakable` — pdfmake wraps every body row in `beginUnbreakableBlock()`/`commitUnbreakableBlock()`, and `commitUnbreakableBlock` keeps only `context.pages[0]`. A row taller than the page body is therefore **silently deleted, not clipped and not moved**. With a 3,279-character line note the whole product row vanished — part number, unit price, quantity and line total — while the totals block still printed a TOTAL that included it.

The product table therefore keeps its original definition and breaks between rows, which is ugly at a break but lossless:

```
      table: { headerRows: 1, widths: ['*', 'auto', 'auto', 'auto', 'auto'], body: tableBody },
```

⚠️ **Never mark the product table `unbreakable` either**, and treat "compressed or clipped" as the wrong mental model wherever it appears: pdfmake **drops** the overflow. The `PRODUCTS & PRICING` eyebrow is deliberately left unattached to the table for the same reason.

⚠️ The underlying exposure is unbounded input: the per-line note is a `<textarea>` with no `maxlength`, and there are no `maxlength` attributes anywhere in the file. Bounding it is a **product decision for the owner**, not something to slip in here.

## Task 6: A repeating header in the browser-print path (workstream B)

**Read this before starting.** This is the only genuinely uncertain task in the plan.

Today `.doc-footer` repeats because it is `position:fixed`, and `.doc-header` does not repeat because it is a normal block. The obvious fix — reserving space with a non-zero `@page` margin — **conflicts with a deliberate decision already in the file**: `@page{margin:0}` is what suppresses Chrome's own URL and date lines. Restoring a margin brings them back on a customer document.

So the approach is a print-only DOM wrap: on `beforeprint`, clone the header into a `<thead>`; browsers repeat a `thead` on every printed page *and* reserve space for it. On `afterprint`, undo it. The screen DOM is untouched.

```js
// ===== PRINT LAYOUT =====
// Browsers repeat a <thead> on every printed page and reserve space for it, which is
// the only reliable way to get a running header without giving @page a margin — and a
// non-zero @page margin would bring back Chrome's own URL/date lines on a customer
// document. Applied on beforeprint, removed on afterprint, so the screen DOM never
// carries the wrapper.
function wrapDocForPrint(rootId) {
  const root = document.getElementById(rootId);
  if (!root || root.querySelector('.print-frame')) return;
  const header = root.querySelector('.doc-header');
  if (!header) return;
  const rest = Array.from(root.children).filter(el => el !== header);

  const frame = document.createElement('table');
  frame.className = 'print-frame';
  const thead = document.createElement('thead');
  const headCell = document.createElement('td');
  headCell.appendChild(header.cloneNode(true));
  const headRow = document.createElement('tr');
  headRow.appendChild(headCell);
  thead.appendChild(headRow);

  const bodyCell = document.createElement('td');
  rest.forEach(el => bodyCell.appendChild(el));
  const bodyRow = document.createElement('tr');
  bodyRow.appendChild(bodyCell);
  const tbody = document.createElement('tbody');
  tbody.appendChild(bodyRow);

  frame.appendChild(thead);
  frame.appendChild(tbody);
  header.style.display = 'none';
  root.appendChild(frame);
}

function unwrapDocAfterPrint(rootId) {
  const root = document.getElementById(rootId);
  if (!root) return;
  const frame = root.querySelector('.print-frame');
  if (!frame) return;
  const bodyCell = frame.querySelector('tbody td');
  Array.from(bodyCell.children).forEach(el => root.appendChild(el));
  frame.remove();
  const header = root.querySelector('.doc-header');
  if (header) header.style.display = '';
}

window.addEventListener('beforeprint', () => {
  wrapDocForPrint('docRoot');
  wrapDocForPrint('plDocRoot');
});
window.addEventListener('afterprint', () => {
  unwrapDocAfterPrint('docRoot');
  unwrapDocAfterPrint('plDocRoot');
});
```

Add to the `@media print` block, immediately before its closing brace:

```css
    .print-frame > thead{display:table-header-group;}
    .print-frame td{padding:0;border:0;}
```
